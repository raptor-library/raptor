# Create a raptorExternalTargets.cmake file.
# This will log the imported targets
set(RAPTOR_EXTERNAL_TARGETS "")
foreach(pkg IN ITEMS
        "raptor::hypre|WITH_HYPRE|HYPRE_INCLUDE_DIRS|HYPRE_LIBRARIES"
        "raptor::petsc|WITH_PETSC|PETSC_INCLUDES|PETSC_LIBRARIES"
        "raptor::ptscotch|WITH_PTSCOTCH|PTSCOTCH_INCLUDE_DIRS|PTSCOTCH_LIBRARIES"
        "raptor::parmetis|WITH_PARMETIS|PARMETIS_INCLUDE_DIRS|PARMETIS_LIBRARIES")
    string(REPLACE "|" ";" pkg "${pkg}")
    list(GET pkg 0 tgt)
    list(GET pkg 1 enabled)
    list(GET pkg 2 incs)
    list(GET pkg 3 libs)
    if (${enabled})
        string(APPEND RAPTOR_EXTERNAL_TARGETS
            "if(NOT TARGET ${tgt})\n"
            "  add_library(${tgt} INTERFACE IMPORTED)\n"
            "  set_target_properties(${tgt} PROPERTIES\n"
            "    INTERFACE_INCLUDE_DIRECTORIES \"${${incs}}\"\n"
            "    INTERFACE_LINK_LIBRARIES \"${${libs}}\")\n"
            "endif()\n")
    endif()
endforeach()
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/raptorExternalTargets.cmake
    "# Generated by RAPtor: external packages found by RAPtor's find modules\n"
    "${RAPTOR_EXTERNAL_TARGETS}")

include(CMakePackageConfigHelpers)

  ${CMAKE_CURRENT_BINARY_DIR}/raptorConfig.cmake
  INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/raptor)

  ${CMAKE_CURRENT_BINARY_DIR}/raptorConfigVersion.cmake
  VERSION ${PROJECT_VERSION}
  COMPATIBILITY SameMajorVersion)

install(
  FILES ${CMAKE_CURRENT_BINARY_DIR}/raptorConfig.cmake
        ${CMAKE_CURRENT_BINARY_DIR}/raptorConfigVersion.cmake
        ${CMAKE_CURRENT_BINARY_DIR}/raptorExternalTargets.cmake
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/raptor)
